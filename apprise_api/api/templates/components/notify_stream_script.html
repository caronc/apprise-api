'' : line.slice(separator + 1); if (value.startsWith(' ')) value = value.slice(1); if (field === 'event') eventName = value; if (field === 'data') data.push(value); }); if (!data.length) return null; return {name: eventName, data: JSON.parse(data.join('\n'))}; } // Read the live log as it arrives, passing each event to onEvent. // Returns the final "result" or "error" event; throws when there is none. async function readNotifyEventStream(response, onEvent) { const contentType = response.headers.get('Content-Type') || ''; } const reader = response.body.getReader(); const decoder = new TextDecoder('utf-8'); let buffer = ''; let finalEvent = null; // Add new text, then hand on every complete event (they end with a blank line). function consume(finalChunk) { buffer += finalChunk;
